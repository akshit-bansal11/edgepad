# Golden frames for Edgepad protocol v1.
#
# Both test suites (android/ in Kotlin, windows/ in C#) read this file. For every
# line they encode the fields and must get exactly these bytes, then decode the
# bytes and must get exactly these fields. That is what stops the two sides from
# drifting apart: a change to one codec without the other fails a test.
#
# Format:  NAME field field ... = hex bytes
# All multi-byte integers are little-endian. The first byte is the frame type;
# the payload length is fixed per type, except TEXT, whose second header byte is
# the length of the UTF-8 text that follows (at most 255 bytes).

MOVE -32768 32767 = 10 00 80 ff 7f

ACTION 3 = 20 03
